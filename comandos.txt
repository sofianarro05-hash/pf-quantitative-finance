# ====================================
# Comandos de Python
# ====================================

# crear un ambiente virtual
python3 -m venv .venv 

# activar ambiente virtual
source .venv/bin/activate

# instalar librerías desde archivo requirements
pip3 install -r requirements.txt

# ====================================
# Comandos de Git
# ====================================

# revisar status
git status

# agregar archivo
git add {nombre_archivo} | ejemplo: git add main.py 

# confirmar cambios
git commit -m "backup"

# cargar cambios al repositorio
git push origin main
